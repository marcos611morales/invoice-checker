// src/ports/drive-api.ts (pure)
export const FOLDER_MIME = "application/vnd.google-apps.folder";
export const SHORTCUT_MIME = "application/vnd.google-apps.shortcut";
export const DRIVE_ID_PATTERN = /^[A-Za-z0-9_-]{10,}$/;
export function isDriveId(s: string): boolean;
export interface DriveEntry {
  id: string; name: string; mimeType: string; modifiedTime: string;
  md5Checksum?: string; createdTime?: string; size?: string; parents?: string[]; trashed?: boolean;
}
export interface DriveApi {
  list(q: string): Promise<DriveEntry[]>;                 // follows nextPageToken, pageSize 1000
  getMeta(fileId: string): Promise<DriveEntry>;
  getBytes(fileId: string): Promise<Uint8Array>;           // alt=media
  createFile(args: { name: string; parentId: string; mimeType: string; body?: Uint8Array | string }): Promise<DriveEntry>;
  updateFile(fileId: string, body: Uint8Array | string, mimeType: string): Promise<DriveEntry>;  // media update
  about(): Promise<{ emailAddress: string }>;
}

// src/adapters/drive/tree.ts
export interface XmlTree { files: DriveEntry[]; ignoredShortcuts: number; ignoredNative: number }
export function folderChildrenQuery(folderId: string): string;   // `'${id}' in parents and trashed = false`
export async function listXmlTree(api: DriveApi, rootFolderId: string): Promise<XmlTree>;

// src/adapters/drive/drive-errors.ts
export type DriveErrorCode = "no_configurado" | "autorizacion_caducada" | "no_encontrado" | "sin_permiso"
  | "limite_de_uso" | "sin_espacio" | "red" | "desconocido";
export class DriveError extends Error { readonly code: DriveErrorCode; readonly mensaje: string; readonly retryable: boolean }
export function toDriveError(err: unknown): DriveError;

// src/adapters/drive/credentials.ts (server-only)
export function makeDriveClient(env: { DRIVE_CLIENT_ID?: string; DRIVE_CLIENT_SECRET?: string; DRIVE_REFRESH_TOKEN?: string }): drive_v3.Drive;
